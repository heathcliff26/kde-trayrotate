package main

        "strings"

        "github.com/heathcliff26/kde-trayrotate/pkg/version"
        "github.com/spf13/cobra"
)

func NewRootCommand() *cobra.Command {
        name := strings.ToLower(version.Name)
        cobra.AddTemplateFunc(
                "ProgramName", func() string {
                        return name
                },
        )

        rootCmd := &cobra.Command{
                Use:   name,
                Short: version.Name + " toggle auto rotate from the system tray",
                Run: func(_ *cobra.Command, _ []string) {
                        run()
                },
        }

        rootCmd.AddCommand(
                version.NewCommand(),
        )

        return rootCmd
}

        cmd := NewRootCommand()
        err := cmd.Execute()
        if err != nil {
                os.Exit(1)
        }
}

package main

import (
        "fmt"
        "log/slog"
        "time"

        "fyne.io/systray"
        "github.com/heathcliff26/kde-trayrotate/icons"
        "github.com/heathcliff26/kde-trayrotate/pkg/kscreendoctor"
)

const (
        Title   = "io.github.heathcliff26.kde-trayrotate"
        Tooltip = "Auto Rotate"

        AutoRotateOn  = "on"
        AutoRotateOff = "off"
)

var doctor = kscreendoctor.NewKScreenDoctor()
var exit = make(chan struct{})

func run() {
        systray.Run(onReady, onExit)
}

func onReady() {
        systray.SetTitle(Title)
        systray.SetOnTapped(toggleAutoRotate)

        systraySetDefault()

        mQuit := systray.AddMenuItem("Quit", "Quit the app")

        // Sets the icon of a menu item.
        quitCh := mQuit.ClickedCh
        go func() {
                <-quitCh
                systray.Quit()
        }()

        go func() {
                for {
                        updateSystray()
                        select {
                        case <-time.Tick(1 * time.Second):
                        case <-exit:
                                return
                        }
                }
        }()
}

func onExit() {
        close(exit)
}

func systraySetDefault() {
        systray.SetIcon(icons.Icon)
        systray.SetTooltip(Tooltip)
}

func updateSystray() {
        status, err := doctor.GetStatus()
        if err != nil {
                slog.Error("Failed to get current screen status", "error", err)
                systraySetDefault()
                return
        }

        state, ok := status["1"]
        if !ok {
                slog.Debug("No screens with auto-rotation support found")
                systraySetDefault()
                return
        }

        updateSystrayFromState(state)
}

func updateSystrayFromState(state string) {
        switch state {
        case kscreendoctor.PolicyNever:
                systray.SetIcon(icons.RotationLocked)
                systray.SetTooltip(fmt.Sprintf("%s: %s", Tooltip, AutoRotateOff))
        case kscreendoctor.PolicyAlways:
                systray.SetIcon(icons.RotationUnlocked)
                systray.SetTooltip(fmt.Sprintf("%s: %s", Tooltip, AutoRotateOn))
        default:
                systraySetDefault()
        }
}

func toggleAutoRotate() {
        status, err := doctor.GetStatus()
        if err != nil {
                slog.Error("Failed to get current screen status", "error", err)
                systraySetDefault()
                return
        }

        state, ok := status["1"]
        if !ok {
                slog.Debug("No screens with auto-rotation support found")
                systraySetDefault()
                return
        }

        var newPolicy string
        switch state {
        case kscreendoctor.PolicyNever:
                newPolicy = kscreendoctor.PolicyAlways
        case kscreendoctor.PolicyInTabledMode:
                newPolicy = kscreendoctor.PolicyNever
        case kscreendoctor.PolicyAlways:
                newPolicy = kscreendoctor.PolicyNever
        }
        err = doctor.SetAutoRotatePolicies(status, newPolicy)
        if err != nil {
                slog.Error("Failed to set new auto-rotation policy", "error", err)
                systraySetDefault()
                return
        }

        updateSystrayFromState(newPolicy)
}

package kscreendoctor

import (
        "encoding/json"
        "fmt"
        "log/slog"
        "os/exec"
        "strconv"
)

const (
        autoRotatePolicyTemplate = "output.%s.autoRotatePolicy.%s"

        PolicyNever        = "never"
        PolicyInTabledMode = "inTabletMode"
        PolicyAlways       = "always"
)

type KScreenDoctor struct {
        // Binary is only configurable for testing purposes.
        binary string
}

// Create a new instance for kscreen-doctor commands
func NewKScreenDoctor() *KScreenDoctor {
        return &KScreenDoctor{
                binary: "kscreen-doctor",
        }
}

// Get the current status of the screen rotation settings
func (ksc *KScreenDoctor) GetStatus() (map[string]string, error) {
        out, err := ksc.runCommand("--json")
        if err != nil {
                return nil, fmt.Errorf("failed to get settings from kscreen-doctor: %w", err)
        }

        var settings ScreenSettings
        err = json.Unmarshal([]byte(out), &settings)
        if err != nil {
                return nil, fmt.Errorf("failed to parse settings from kscreen-doctor: %w", err)
        }

        if len(settings.Outputs) == 0 {
                return nil, fmt.Errorf("no outputs found")
        }

        status := make(map[string]string)

        for i, output := range settings.Outputs {
                if output.AutoRotatePolicy != nil {
                        policy := ""
                        switch *output.AutoRotatePolicy {
                        case 0:
                                policy = PolicyNever
                        case 1:
                                policy = PolicyInTabledMode
                        case 2:
                                policy = PolicyAlways
                        default:
                                slog.Debug("Unknown auto-rotation state", slog.Int("output", i), slog.Int("state", *output.AutoRotatePolicy))
                        }
                        // The screen index used by kscreen-doctor counts from 1 -> i+1
                        status[strconv.Itoa(i+1)] = policy
                }
        }

        return status, nil
}

// Set the auto-rotation policy for a given output
func (ksc *KScreenDoctor) SetAutoRotatePolicies(outputs map[string]string, policy string) error {
        for output, state := range outputs {
                if state == policy {
                        continue
                }

                _, err := ksc.runCommand(fmt.Sprintf(autoRotatePolicyTemplate, output, policy))
                if err != nil {
                        return fmt.Errorf("failed to set auto-rotation policy for output %s: %w", output, err)
                }
        }
        return nil
}

func (ksc *KScreenDoctor) runCommand(args ...string) (string, error) {
        cmd := exec.Command(ksc.binary, args...)
        out, err := cmd.Output()
        if err != nil {
                return "", err
        }
        return string(out), nil
}

package version

import (
        "runtime"
        "runtime/debug"
        "strings"

        "github.com/spf13/cobra"
)

const (
        Name = "kde-trayrotate"
)

// NOTE: The $Format strings are replaced during 'git archive' thanks to the
// companion .gitattributes file containing 'export-subst' in this same
// directory.  See also https://git-scm.com/docs/gitattributes
var gitCommit string = "$Format:%H$" // sha1 from git, output of $(git rev-parse HEAD)
var gitVersion string = ""

func init() {
        initGitCommit()
        initGitVersion()
}

func initGitCommit() {
        if strings.HasPrefix(gitCommit, "$Format") {
                var commit string
                buildinfo, _ := debug.ReadBuildInfo()
                for _, item := range buildinfo.Settings {
                        if item.Key == "vcs.revision" {
                                commit = item.Value
                                break
                        }
                }
                if commit == "" {
                        commit = "Unknown"
                }
                gitCommit = commit
        }
        if gitCommit == "" {
                gitCommit = "Unknown"
        }
        if len(gitCommit) > 7 {
                gitCommit = gitCommit[:7]
        }
}

func initGitVersion() {
        if gitVersion == "" {
                buildinfo, _ := debug.ReadBuildInfo()
                gitVersion = buildinfo.Main.Version
        }
}

// Create a new version command with the given app name
func NewCommand() *cobra.Command {
        cmd := &cobra.Command{
                Use:   "version",
                Short: "Print version information and exit",
                Run: func(cmd *cobra.Command, args []string) {
                        cmd.Print(VersionInfoString())
                },
        }
        // Override to prevent parent function from running
        cmd.PersistentPreRun = func(cmd *cobra.Command, args []string) {}

        return cmd
}

// Return the version string
func Version() string {
        return gitVersion
}

// Return the git commit hash
func Commit() string {
        return gitCommit
}

// Return a formated string containing the version, git commit and go version the app was compiled with.
func VersionInfoString() string {
        result := Name + ":\n"
        result += "    Version: " + gitVersion + "\n"
        result += "    Commit:  " + gitCommit + "\n"
        result += "    Go:      " + runtime.Version() + "\n"

        return result
}
